use std::fmt;

#[derive(Debug, Clone, PartialEq, Eq, Hash)]
pub struct StateName(String);

impl StateName {
    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl TryFrom<&str> for StateName {
    type Error = StateError;

    fn try_from(value: &str) -> Result<Self, Self::Error> {
        if value.is_empty() {
            Err(StateError::EmptyState)
        } else {
            Ok(Self(value.to_string()))
        }
    }
}

impl TryFrom<String> for StateName {
    type Error = StateError;

    fn try_from(value: String) -> Result<Self, Self::Error> {
        if value.is_empty() {
            Err(StateError::EmptyState)
        } else {
            Ok(Self(value))
        }
    }
}

impl fmt::Display for StateName {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        self.0.fmt(f)
    }
}

pub fn try_allow(
    mut self,
    from: impl TryInto<StateName, Error = StateError>,
    to: impl TryInto<StateName, Error = StateError>,
) -> Result<Self, StateError> {
    let from = from.try_into()?;
    let to = to.try_into()?;

    if from == to {
        return Err(StateError::SelfTransition {
            state: from.to_string(),
        });
    }

    self.transitions
        .entry(from)
        .or_default()
        .insert(to);

    Ok(self)
}
